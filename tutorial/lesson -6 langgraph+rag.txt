Step 1 — Complete practical code
LangGraph RAG Retry Cycle
import {
  END,
  START,
  StateGraph,
  StateSchema,
} from "@langchain/langgraph";

import z from "zod";

export const ragCycle = async () => {

  // =========================================
  // 1. STATE
  // =========================================

  const State = new StateSchema({
    question: z.string(),

    searchQuery: z.string(),

    documents: z.array(z.string()),

    retryCount: z.number(),

    answer: z.string(),
  });


  // =========================================
  // 2. REWRITE QUERY NODE
  // =========================================

  const rewriteQueryNode = async (state) => {

    console.log("🔄 Rewrite Query");

    return {
      searchQuery: state.question + " detailed explanation",
    };
  };


  // =========================================
  // 3. RETRIEVE NODE
  // =========================================

  const retrieveNode = async (state) => {

    console.log("🔍 Retrieve Documents");

    // Real project me yaha Qdrant retriever call hoga

    if (state.retryCount === 0) {
      return {
        documents: [],
      };
    }

    return {
      documents: [
        "RAG stands for Retrieval Augmented Generation.",
        "RAG retrieves relevant documents before generating an answer.",
      ],
    };
  };


  // =========================================
  // 4. CHECK DOCUMENTS
  // =========================================

  const checkDocumentsNode = async (state) => {

    console.log(
      "📄 Documents found:",
      state.documents.length
    );

    return {};
  };


  // =========================================
  // 5. ROUTER
  // =========================================

  const routeAfterDocumentCheck = (state) => {

    if (state.documents.length > 0) {

      return "answer";

    }

    if (state.retryCount >= 2) {

      return "end";

    }

    return "retry";
  };


  // =========================================
  // 6. RETRY COUNT NODE
  // =========================================

  const retryNode = async (state) => {

    console.log("♻️ Retry");

    return {
      retryCount: state.retryCount + 1,
    };
  };


  // =========================================
  // 7. ANSWER NODE
  // =========================================

  const generateAnswerNode = async (state) => {

    console.log("🤖 Generate Answer");

    return {
      answer:
        "Based on the retrieved documents: " +
        state.documents.join(" "),
    };
  };


  // =========================================
  // 8. GRAPH
  // =========================================

  const workflow = new StateGraph(State)

    // Nodes
    .addNode("rewriteQuery", rewriteQueryNode)

    .addNode("retrieve", retrieveNode)

    .addNode("checkDocuments", checkDocumentsNode)

    .addNode("retry", retryNode)

    .addNode("generateAnswer", generateAnswerNode)


    // START
    .addEdge(
      START,
      "rewriteQuery"
    )


    // Rewrite → Retrieve
    .addEdge(
      "rewriteQuery",
      "retrieve"
    )


    // Retrieve → Check
    .addEdge(
      "retrieve",
      "checkDocuments"
    )


    // Check → Router
    .addConditionalEdges(
      "checkDocuments",
      routeAfterDocumentCheck,
      {
        answer: "generateAnswer",

        retry: "retry",

        end: END,
      }
    )


    // Retry → Rewrite
    .addEdge(
      "retry",
      "rewriteQuery"
    )


    // Answer → END
    .addEdge(
      "generateAnswer",
      END
    );


  // =========================================
  // 9. COMPILE
  // =========================================

  const app = workflow.compile();


  // =========================================
  // 10. INVOKE
  // =========================================

  const result = await app.invoke({

    question: "What is RAG?",

    searchQuery: "",

    documents: [],

    retryCount: 0,

    answer: "",
  });


  console.log("\nFINAL RESULT:");
  console.log(result);
};


ragCycle();

Real system me iske saath timeout, error handling, logging/observability aur fallback response bhi rakhoge.

Interview me ise kaise explain karogi?

"I use LangGraph conditional edges to implement a self-correcting RAG workflow. After retrieval, I check whether relevant documents were found. If documents are available, the workflow proceeds to reranking and answer generation. If no documents are found, it rewrites the query and performs retrieval again. I also maintain a retry counter and maximum retry limit to prevent infinite loops."

Ye answer tumhare LangChain + Qdrant + RAG + LangGraph project ko production perspective se explain karne ke liye kaafi strong hai.
